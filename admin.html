// ========== RENDER REQUESTS ========== function renderRequests() { const tbody = document.getElementById("requests-body"); const search = normalize(document.getElementById("request-search").value); const filtered = requestRows.filter(item => { const waste = Array.isArray(item.selectedWasteTypes) ? item.selectedWasteTypes.join(" ") : ""; const text = [ item.requestId, item.email, item.phone, item.category, item.address, item.status, waste ].join(" "); return normalize(text).includes(search); }); if (!filtered.length) { renderEmpty(tbody, 8); return; } tbody.innerHTML = filtered .map(item => { const wasteTypes = Array.isArray(item.selectedWasteTypes) ? item.selectedWasteTypes : []; const locationText = item.locationShared ? `${Number(item.latitude || 0).toFixed(5)}, ${Number(item.longitude || 0).toFixed(5)}` : "Not shared"; return ` ${escapeHtml(item.requestId || item.id)}
${escapeHtml(item.id)}
${escapeHtml(item.email || "—")}
${escapeHtml(item.phone || "—")}
${escapeHtml(item.category || "—")}
${wasteTypes.length ? wasteTypes.map(w => `${escapeHtml(w)}`).join("") : "—"}
${escapeHtml(item.address || "—")} ${escapeHtml(locationText)} ${escapeHtml(item.status || "—")} ${escapeHtml(formatDate(item.createdAt))} `; }) .join(""); } // ========== SEARCH ========== document.getElementById("request-search").addEventListener("input", renderRequests); // ========== REFRESH ========== document.querySelectorAll("[data-refresh]").forEach(button => { button.addEventListener("click", async () => { button.disabled = true; try { if (button.dataset.refresh === "requests") await loadRequests(); } finally { button.disabled = false; } }); });
